CREATE LOGIN ksef_reader WITH PASSWORD = 'SilneHaslo123!';
-- w kazdej bazie firmowej (CDN_*):
USE CDN_Firma; CREATE USER ksef_reader FOR LOGIN ksef_reader;
ALTER ROLE db_datareader ADD MEMBER ksef_reader;
